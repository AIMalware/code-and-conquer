560. Subarray Sum Equals KGiven an array of integers nums and an integer k, return the total number of subarrays whose sum equals to k.

A subarray is a contiguous non-empty sequence of elements within an array.

 

Example 1:

Input: nums = [1,1,1], k = 2
Output: 2
Example 2:

Input: nums = [1,2,3], k = 3
Output: 2

approach=formulae=pdum-K
2.add hashmap f into the cnt 
3.increase cnt  of hash map
4.return cnt 


class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
        unordered_map<int,int> mp;
        int psum=0;
        int cnt=0;
        //unordered_map<int,int> mp;
        mp[0]=1;
        for(int i=0;i<nums.size();i++)
        {
            psum+=nums[i];
            int r=psum-k;
            cnt+=mp[r];
            mp[psum]++;
        }
        return cnt;
        
        
    }
};
